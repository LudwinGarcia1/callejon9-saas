# CAL-121 · el atributo pattern del identificador en un navegador real (2026-10-22 11:24 America/Mexico_City)
# Microsoft Edge 155 en modo headless, con una página mínima que tiene el pattern actual de signup-view.tsx
# y el mismo pattern con el guion escapado. Se le asigna cada valor al campo y se lee validity.patternMismatch.
$ msedge --headless=new --disable-gpu --enable-logging=stderr --v=0 --dump-dom file:///…/slug-pattern.html

## Página usada
<!doctype html><meta charset="utf-8"><title>pattern</title>
<input id="actual" pattern="[a-z0-9-]{3,80}">
<input id="escapado" pattern="[a-z0-9\-]{3,80}">
<pre id="out"></pre>
<script>
const rows = [];
for (const id of ["actual", "escapado"]) {
  const el = document.getElementById(id);
  for (const v of ["la-esquina", "La_Esquina", "ab"]) {
    el.value = v;
    rows.push(`${id} pattern="${el.pattern}" valor="${v}" patternMismatch=${el.validity.patternMismatch} valido=${el.checkValidity()}`);
  }
}
document.getElementById("out").textContent = navigator.userAgent + "\n" + rows.join("\n");
</script>
